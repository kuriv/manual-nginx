流量控制

流量控制

限制单个客户端同时建立的活跃连接数量，主要用于防止单用户通过多线程下载、多连接占用耗尽服务器连接资源。修改 nginx.conf 配置文件内容如下。

user                nginx;
worker_processes    auto;

events {
    worker_connections  1024;
}

http {
    include         mime.types;
    default_type    application/octet-stream;

    limit_conn_zone $binary_remote_addr zone=ip_conn:10m;

    server {
        listen      80;
        server_name localhost;
        root        /usr/share/nginx/html;

        location /download {
            limit_conn ip_conn  1;
            limit_conn_status   429;
            try_files           /download.html =404;
        }
    }
}

基于漏桶算法限制单位时间内的请求数量，是生产环境中最常用的限流手段，主要用于防 CC 攻击、防爬虫、防接口被高频恶意调用。修改 nginx.conf 配置文件内容如下。

user                nginx;
worker_processes    auto;

events {
    worker_connections  1024;
}

http {
    include         mime.types;
    default_type    application/octet-stream;

    limit_req_zone $binary_remote_addr zone=ip_req:10m rate=1r/s;

    server {
        listen      80;
        server_name localhost;
        root        /usr/share/nginx/html;

        location /api {
            limit_req zone=ip_req burst=2 nodelay;
            limit_req_status    429;
            try_files           /api.html =404;
        }
    }
}

限制单个连接的下载或传输速度，从而控制出口带宽占用。修改 nginx.conf 配置文件内容如下。

user                nginx;
worker_processes    auto;

events {
    worker_connections  1024;
}

http {
    include         mime.types;
    default_type    application/octet-stream;

    limit_conn_zone $binary_remote_addr zone=ip_conn:10m;

    server {
        listen      80;
        server_name localhost;
        root        /usr/share/nginx/html;

        location /download {
            limit_conn ip_conn  1;
            limit_conn_status   429;
            limit_rate_after    1m;
            limit_rate          100k;
            try_files           /download.html =404;
        }
    }
}

允许或限制指定单个 IP 或网段访问相关资源。修改 nginx.conf 配置文件内容如下。

user                nginx;
worker_processes    auto;

events {
    worker_connections  1024;
}

http {
    include         mime.types;
    default_type    application/octet-stream;

    server {
        listen      80;
        server_name localhost;
        root        /usr/share/nginx/html;

        location /admin {
            allow       127.0.0.1;
            allow       192.168.0.0/16;
            deny        all;
            try_files   /admin.html =404;
        }

        location /user {
            deny        123.45.67.89;
            allow       all;
            try_files   /user.html =404;
        }
    }
}
